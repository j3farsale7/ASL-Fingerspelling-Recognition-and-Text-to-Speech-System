                # تحويل الصورة إلى تدرجات الرمادي

                # تغيير حجم الصورة إلى 50x50 بكسل، حيث نحتاج إلى حجم ثابت لجميع الصور في مجموعة البيانات
                data.append(resized)  # إضافة الصورة المعالجة إلى قائمة البيانات
                #print(data)
                target.append(label_classes_dict[classes])  # إضافة التصنيف المقابل للصورة إلى قائمة الأهداف
            except Exception as e:  # التعامل مع الأخطاء المحتملة أثناء معالجة الصورة
                print('Exception:', e)  # طباعة رسالة الخطأ

from keras.models import Sequential  # استيراد النموذج Sequential من مكتبة Keras
from keras.layers import Convolution2D  # استيراد طبقة Convolution2D (التلافيفية) من مكتبة Keras
from keras.layers import MaxPooling2D  # استيراد طبقة MaxPooling2D (التجمع الأقصى) من مكتبة Keras
from keras.layers import Flatten  # استيراد طبقة Flatten (التسطيح) من مكتبة Keras
from keras.layers import Dense, Dropout  # استيراد طبقتي Dense (الكثيفة) و Dropout (التسرب) من مكتبة Keras
import os  # استيراد مكتبة os للتعامل مع نظام الملفات
os.environ["CUDA_VISIBLE_DEVICES"] = "1"  # تعيين الجهاز المرئي لوحدة معالجة الرسوميات (

GPU) إلى 1
sz = 128  # تعيين حجم الصور إلى 128x128 بكسل

# خطوة 1 - بناء الشبكة العصبية الالتفافية (CNN)

# تهيئة الشبكة العصبية
classifier = Sequential()  # إنشاء نموذج Sequential

# إضافة الطبقة الالتفافية الأولى وطبقة التجمع الأقصى
classifier.add(Convolution2D(32, (3, 3), input_shape=(sz, sz, 1), activation='relu'))  # إضافة طبقة Convolution2D مع 32 مرشحًا وحجم مرشح 3x3
classifier.add(MaxPooling2D(pool_size=(2, 2)))  # إضافة طبقة MaxPooling2D بحجم تجمع 2x2

# إضافة الطبقة الالتفافية الثانية وطبقة التجمع الأقصى
classifier.add(Convolution2D(32, (3, 3), activation='relu'))  # إضافة طبقة Convolution2D مع 32 مرشحًا وحجم مرشح 3x3
# input_shape هي الخرائط المميزة من الطبقة الالتفافية السابقة
classifier.add(MaxPooling2D(pool_size=(2, 2)))  # إضافة طبقة MaxPooling2D بحجم تجمع 2x2
#classifier.add(Convolution2D(32, (3, 3), activation='relu'))
# input_shape هي الخرائط المميزة من الطبقة الالتفافية السابقة
#classifier.add(MaxPooling2D(pool_size=(2, 2)))

# تسطيح الطبقات
classifier.add(Flatten())  # إضافة طبقة Flatten لتحويل الخرائط المميزة إلى شكل مسطح

# إضافة طبقة كثيفة متصلة بالكامل
classifier.add(Dense(units=128, activation='relu'))  # إضافة طبقة Dense مع 128 وحدة وتفعيل 'relu'
classifier.add(Dropout(0.40))  # إضافة طبقة Dropout بنسبة 40% لتقليل الإفراط في التعميم
classifier.add(Dense(units=96, activation='relu'))  # إضافة طبقة Dense مع 96 وحدة وتفعيل 'relu'
classifier.add(Dropout(0.40))  # إضافة طبقة Dropout بنسبة 40% لتقليل الإفراط في التعميم
classifier.add(Dense(units=64, activation='relu'))  # إضافة طبقة Dense مع 64 وحدة وتفعيل 'relu'
classifier.add(Dense(units=27, activation='softmax'))  # إضافة طبقة Dense مع 27 وحدة وتفعيل 'softmax' لتصنيف متعدد الفئات

# تجميع الشبكة العصبية
classifier.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])  # تجميع النموذج مع استخدام 'adam' كمحسن و'categorical_crossentropy' كدالة خسارة
